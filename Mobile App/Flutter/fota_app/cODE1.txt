  /* Save the login time function */
  void LoginTime() {
    setState(() {
      DateTime currentTime = DateTime.now();
      String formattedTime =
          DateFormat('yyyy-MM-dd hh:mm a').format(currentTime);
      _MobileINFO.update({
        "LastLogin": formattedTime,
      });
    });
  }

    try {
      /* Mobile INFO */
      DeviceInfoPlugin deviceInfo = DeviceInfoPlugin();
      AndroidDeviceInfo androidInfo = await deviceInfo.androidInfo;
      //print(androidInfo);
      _MobileINFO.update({
        "UserINFO":
            'Manufacture: ${androidInfo.manufacturer} ---  Model: ${androidInfo.model} ---  SW Version: ${androidInfo.version}',
      });
      if ((androidInfo.hardware == Mobile_HARDWARE_Master) &&
          (androidInfo.fingerprint == Mobile_Fingerprint_Master) &&
          (androidInfo.host == Mobile_HOST_Master)) {
        // print('FingerPrint: ${androidInfo.fingerprint}');
        // print('HOST : ${androidInfo.host}');

      //print('Running on ${androidInfo.manufacturer}');

    } catch (e) {
      //   print('Failed to get device info: $e');
    }

/* Listen */
  void ListenFun() {
    /* Current states */
    _MobileLockApp.onValue.listen((DatabaseEvent event) {
      final dataCurrent = event.snapshot.value;
      if (event.snapshot.exists &&
          dataCurrent != null &&
          dataCurrent is Map<Object?, Object?>) {
        // Update the state with the new software version
        setState(() {
          setState(() {
            g_LockApp = dataCurrent['State'] as bool;
            g_LockAppPassword = dataCurrent['Password'] as String;
          });
        });
      } else {
        print('Data type: ${dataCurrent.runtimeType}');
      }
    });
  }

  /* Check the password */
  void showPasswordDialog() async {
    String? enteredPassword = await showDialog<String>(
      context: context,
      builder: (BuildContext context) {
        String input = "";
        return AlertDialog(
          title: Text('Enter Password'),
          content: TextField(
            keyboardType: TextInputType.number,
            obscureText: true,
            onChanged: (value) {
              input = value;
            },
          ),
          actions: <Widget>[
            TextButton(
              child: Text('Cancel'),
              onPressed: () {
                Navigator.of(context).pop();
              },
            ),
            TextButton(
              child: Text('OK'),
              onPressed: () {
                Navigator.of(context).pop(input);
              },
            ),
          ],
        );
      },
    );

    if (enteredPassword != null) {
      // Compare the entered password with the correct password
      if (enteredPassword == g_LockAppPassword) {
        await _MobileLockApp.update({
          'State': !g_LockApp,
        });
        // Password is correct
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              !g_LockApp
                  ? 'Unlocked: Anyone can access your vehicle'
                  : 'Locked: Your vehicle is secured',
              style: TextStyle(
                color: Colors.white,
              ),
            ),
            backgroundColor: Colors.white12,
          ),
        );
      } else {
        // Password is incorrect
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              'Password is incorrect',
              style: TextStyle(
                color: Colors.white,
              ),
            ),
            backgroundColor: Colors.white12,
          ),
        );
      }
    }
  }
